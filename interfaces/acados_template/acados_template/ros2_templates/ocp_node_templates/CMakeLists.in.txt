{%- if solver_options.qp_solver %}
    {%- set qp_solver = solver_options.qp_solver %}
{%- else %}
    {%- set qp_solver = "FULL_CONDENSING_HPIPM" %}
{%- endif -%}

{%- set qp_lib = "" %}
{%- set qp_include = "" %}
{%- if "QPOASES" in qp_solver %}
    {%- set qp_lib = "qpOASES_e" %}
    {%- set qp_include = "qpOASES_e" %}
{%- elif "DAQP" in qp_solver %}
    {%- set qp_lib = "daqp" %}
    {%- set qp_include = "daqp/include" %}
{%- elif "OSQP" in qp_solver %}
    {%- set qp_lib = "osqp" %}
    {%- set qp_include = "osqp" %}
{%- elif "CLARABEL" in qp_solver %}
    {%- set qp_lib = "clarabel_c" %}
{%- elif "QPDUNES" in qp_solver %}
    {%- set qp_lib = "qpdunes" %}
    {%- set qp_include = "qpdunes/include" %}
{%- elif "HPMPC" in qp_solver %}
    {%- set qp_lib = "hpmpc" %}
    {%- set qp_include = "hpmpc/include" %}
{%- elif "OOQP" in qp_solver %}
    {%- set qp_lib = "ooqp" %}
{%- endif -%}

cmake_minimum_required(VERSION 3.16)
project({{ ros_opts.package_name }})
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.27)
    cmake_policy(SET CMP0148 OLD)
endif()

set(ACADOS_WARN_FLAGS)
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(ACADOS_WARN_FLAGS -Wall -Wextra -Wpedantic)
endif()

# --- ROS DEPENDENCIES ---
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package({{ ros_opts.package_name }}_interface REQUIRED)
{%- if solver_options.with_batch_functionality %}
find_package(OpenMP REQUIRED)
{%- endif %}

# --- ACADOS PATHS ---
set(ACADOS_INCLUDE_PATH {{ code_gen_options.acados_include_path }})
set(ACADOS_LIB_DIR {{ code_gen_options.acados_lib_path }})
set(ACADOS_GENERATED_CODE_DIR {{ code_gen_options.code_export_directory }})
{%- if qp_lib != "" %}
find_library(ACADOS_QP_SOLVER_LIB
    NAMES {{ qp_lib }}
    PATHS ${ACADOS_LIB_DIR}
    NO_DEFAULT_PATH
    REQUIRED
)
{%- endif %}

# --- NODE LIBRARY ---
add_library({{ ros_opts.node_name }}_lib STATIC
    src/node.cpp
)

# --- TARGET CONFIGURATION ---
target_compile_options({{ ros_opts.node_name }}_lib PRIVATE ${ACADOS_WARN_FLAGS})
target_include_directories({{ ros_opts.node_name }}_lib PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
    ${ACADOS_GENERATED_CODE_DIR}
    ${ACADOS_INCLUDE_PATH}
    ${ACADOS_INCLUDE_PATH}/acados
    ${ACADOS_INCLUDE_PATH}/blasfeo/include
    ${ACADOS_INCLUDE_PATH}/hpipm/include
    {%- if qp_include != "" %}
    ${ACADOS_INCLUDE_PATH}/{{ qp_include }}
    {%- endif %}
)
target_link_libraries({{ ros_opts.node_name }}_lib
    ${ACADOS_GENERATED_CODE_DIR}/libacados_ocp_solver_{{ name }}{{ code_gen_options.shared_lib_ext }}
    ${ACADOS_LIB_DIR}/libacados.so
    ${ACADOS_LIB_DIR}/libblasfeo.so
    ${ACADOS_LIB_DIR}/libhpipm.so
    m
    {%- if qp_lib != "" %}
    ${ACADOS_QP_SOLVER_LIB}
    {%- endif %}
    {%- if solver_options.with_batch_functionality %}
    OpenMP::OpenMP_CXX
    {%- endif %}
)

# --- DEPENDENCIES ---
if(COMMAND ament_target_dependencies)
    ament_target_dependencies({{ ros_opts.node_name }}_lib
        rclcpp
        std_msgs
        {{ ros_opts.package_name }}_interface
    )
else()
    target_link_libraries({{ ros_opts.node_name }}_lib
        rclcpp::rclcpp
        std_msgs::std_msgs
        {{ ros_opts.package_name }}_interface::{{ ros_opts.package_name }}_interface
    )
endif()

add_executable({{ ros_opts.node_name }} 
    src/main.cpp
)
target_compile_options({{ ros_opts.node_name }} PRIVATE
    ${ACADOS_WARN_FLAGS}
)
target_link_libraries({{ ros_opts.node_name }}
    {{ ros_opts.node_name }}_lib
)

# --- INSTALLATIONS ---
install(TARGETS 
    {{ ros_opts.node_name }}
    RUNTIME DESTINATION lib/${PROJECT_NAME}
)

install(DIRECTORY 
    include/
    DESTINATION include
)

install(FILES 
    ${ACADOS_GENERATED_CODE_DIR}/libacados_ocp_solver_{{ name }}{{ code_gen_options.shared_lib_ext }}
    DESTINATION lib
)

# --- EXPORT HOOKS ---
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/env-hooks/export_acados_path.sh.in")

# --- TESTS ---
if(BUILD_TESTING)
    find_package(launch_testing_ament_cmake REQUIRED)
    add_launch_test(
        test/test_{{ ros_opts.package_name }}.launch.py
        TARGET test_{{ ros_opts.package_name }}
        TIMEOUT 180
    )

    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(gtest_{{ ros_opts.package_name }}
        test/test_{{ ros_opts.package_name }}.cpp
    )
    if(TARGET gtest_{{ ros_opts.package_name }})
        target_compile_options(gtest_{{ ros_opts.package_name }} PRIVATE
            ${ACADOS_WARN_FLAGS}
        )
        target_link_libraries(gtest_{{ ros_opts.package_name }}
            {{ ros_opts.node_name }}_lib
        )
    endif()
endif()

ament_package()
